  return new Promise((resolve) => {
    db.transaction((tx) => {
      tx.executeSql(`CREATE TABLE IF NOT EXISTS transactions (
        id TEXT PRIMARY KEY NOT NULL,
        user_id TEXT,
        account_id TEXT,
        category_id TEXT,
        amount REAL NOT NULL,
        currency TEXT NOT NULL,
        converted_amount REAL,
        type TEXT NOT NULL CHECK (type IN ('income','expense')),
        note TEXT,
        is_recurring INTEGER DEFAULT 0,
        occurred_at TEXT DEFAULT (datetime('now')),
        deleted INTEGER DEFAULT 0,
        synced INTEGER DEFAULT 0,
        updated_at TEXT DEFAULT (datetime('now'))

        id TEXT PRIMARY KEY NOT NULL,

src/services/syncService.js
import { supabase } from '../config/supabase';
import db from './database';

/**
 * Synchronisation bidirectionnelle :
 * 1. Push des lignes locales non synchronisées (synced = 0)
 * 2. Pull des changements distants (updated_at plus récent)
 */
export async function syncAll(userId) {
  try {
    // ---- PUSH local → cloud ----
    await pushTable('transactions', userId);
    await pushTable('accounts', userId);
    await pushTable('categories', userId);

    // ---- PULL cloud → local ----
    await pullTable('transactions', userId);
    await pullTable('accounts', userId);
    await pullTable('categories', userId);

    return true;
  } catch (e) {
    console.warn('[NKAP Sync] erreur:', e.message);
    return false;
  }
}

async function pushTable(table, userId) {
  const rows = await new Promise((res) =>
    db.transaction((tx) =>
      tx.executeSql(
        `SELECT * FROM ${table} WHERE user_id=? AND synced=0`,
        [userId], (_, { rows }) => res(rows._array)
      )
    )
  );

  for (const row of rows) {
    const payload = { ...row };
    delete payload.synced;
    payload.updated_at = new Date().toISOString();
    payload.deleted = !!row.deleted;

    const { error } = await supabase.from(table).upsert(payload);
    if (!error) markSynced(table, row.id);
  }
}

function markSynced(table, id) {
  db.transaction((tx) =>
    tx.executeSql(`UPDATE ${table} SET synced=1 WHERE id=?`, [id])
  );
}

async function pullTable(table, userId) {
  const { data, error } = await supabase.from(table).select('*').eq('user_id', userId);
  if (error) throw error;

  for (const row of data) {
    await new Promise((res) =>
      db.transaction((tx) => {
        const cols = Object.keys(row).filter((k) => k !== 'synced');
        const vals = cols.map((k) => row[k]);
        const updates = cols.map((c) => `${c}=?`).join(',');
        tx.executeSql(
          `INSERT OR REPLACE INTO table({table} (table({cols.join(',')},synced) VALUES (${'?'.repeat(cols.length)},1)`.replace('VALUES (' , 'VALUES (' ),
          [...vals, 1]
        );
        res();
      })
    );
  }
}

const BASE_URL = 'https://api.exchangerate-api.com/v4/latest';
const CACHE_KEY = 'nkap_rates_cache';
let cache = null;

export async function getRates(baseCurrency = 'USD') {
  if (cache && cache.base === baseCurrency &&
      Date.now() - cache.timestamp < 60 * 60 * 1000) {
    return cache.rates; // cache 1h
  }
  const res = await fetch(`BASEURL/{BASE_URL}/BASEU​RL/{baseCurrency}`);
  const json = await res.json();
  cache = { base: baseCurrency, rates: json.rates, timestamp: Date.now() };
  return json.rates;
}

export async function convert(amount, from, to) {
  if (from === to) return amount;
  const rates = await getRates(from);
  return amount * (rates[to] || 1);
}

import * as LocalAuthentication from 'expo-local-authentication';

export async function isBiometricAvailable() {
  const hasHardware = await LocalAuthentication.hasHardwareAsync();
  const enrolled = await LocalAuthentication.isEnrolledAsync();
  return hasHardware && enrolled;
}

export async function getBiometricType() {
  const types = await LocalAuthentication.supportedAuthenticationTypesAsync();
  if (types.includes(LocalAuthentication.AuthenticationType.FACIAL_RECOGNITION))
    return 'Face ID';
  if (types.includes(LocalAuthentication.AuthenticationType.FINGERPRINT))
    return 'Empreinte';
  return 'Biométrie';
}

export async function authenticateBiometric(prompt = 'Déverrouillez Nkap') {
  const result = await LocalAuthentication.authenticateAsync({
    promptMessage: prompt,
    fallbackLabel: 'Utiliser le code PIN',
  });
  return result.success;
}

import React, { createContext, useContext, useEffect, useState } from 'react';
import { supabase } from '../config/supabase';
import { initDB } from '../services/database';
import { syncAll } from '../services/syncService';

const AuthContext = createContext({});
export const useAuth = () => useContext(AuthContext);

export function AuthProvider({ children }) {
  const [session, setSession] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    initDB().then(() =>
      supabase.auth.getSession().then(({ data: { session } }) => {
        setSession(session);
        setLoading(false);
      })
    );

    const { data: listener } = supabase.auth.onAuthStateChange((_e, s) => {
      setSession(s);
      if (s) ensureProfile(s.user);
    });
    return () => listener?.subscription.unsubscribe();
  }, []);

  async function ensureProfile(user) {
    await supabase.from('profiles').upsert({ id: user.id, email: user.email });
    await syncAll(user.id); // sync au login
  }

  const signInEmail = (email, password) =>
    supabase.auth.signInWithPassword({ email, password });

  const signUpEmail = (email, password) =>
    supabase.auth.signUp({ email, password });

  const signInGoogle = async () => {
    await supabase.auth.signInWithOAuth({
      provider: 'google',
      options: { redirectTo: 'nkap://auth/callback' },
    });
  };

  const signOut = () => supabase.auth.signOut();

  return (
    <AuthContext.Provider value={{ session, loading, signInEmail, signUpEmail, signInGoogle, signOut }}>
      {children}
    </AuthContext.Provider>
  );
}

Sync incrémentale avec horodatage, résolution de conflits (last-write-wins), opérations par lots, et gestion des suppressions douces.

src/services/syncService.js
import { supabase } from '../config/supabase';
import db from './database';
import AsyncStorage from '@react-native-async-storage/async-storage';

const TABLES = ['accounts', 'categories', 'transactions'];
const LAST_PULL_KEY = 'nkap_last_pull_at';

// ============================================================
//  POINT D'ENTRÉE
// ============================================================

/**
 * Synchronisation bidirectionnelle incrémentale :
 * 1. PUSH : envoie les lignes locales modifiées (synced = 0)
 * 2. PULL : récupère uniquement les lignes distantes modifiées
 *    depuis le dernier pull (updated_at > last_pulled_at)
 *
 * Résolution de conflits : LAST-WRITE-WINS
 * → la version avec le updated_at le plus récent gagne.
 */
export async function syncAll(userId) {
  const startedAt = new Date().toISOString();

  try {
    // ---- Phase 1 : PUSH local → cloud ----
    for (const table of TABLES) {
      await pushTable(table, userId);
    }

    // ---- Phase 2 : PULL cloud → local (incrémental) ----
    const lastPull = await AsyncStorage.getItem(LAST_PULL_KEY);
    for (const table of TABLES) {
      await pullTable(table, userId, lastPull);
    }

    // ---- Phase 3 : purge locale des lignes supprimées distantes ----
    await purgeDeletedLocally(userId);

    // ---- Mise à jour du curseur de synchronisation ----
    await AsyncStorage.setItem(LAST_PULL_KEY, startedAt);

    return true;
  } catch (e) {
    console.warn('[NKAP Sync] erreur:', e.message);
    return false;
  }
}

// ============================================================
//  PUSH — Local → Cloud
// ============================================================

async function pushTable(table, userId) {
  const rows = await queryLocal(
    `SELECT * FROM ${table} WHERE user_id=? AND synced=0`,
    [userId]
  );

  if (rows.length === 0) return;

  // Traitement par lots de 50 (limite raisonnable pour Supabase)
  const BATCH_SIZE = 50;
  for (let i = 0; i < rows.length; i += BATCH_SIZE) {
    const batch = rows.slice(i, i + BATCH_SIZE).map((row) => {
      const payload = { ...row };
      delete payload.synced;
      payload.deleted = !!Number(row.deleted);
      payload.updated_at = new Date().toISOString();
      return payload;
    });

    const { error } = await supabase.from(table).upsert(batch, {
      onConflict: 'id',
    });

    if (error) {
      console.warn(`[NKAP Sync] push ${table}:`, error.message);
      continue; // on tente le lot suivant au prochain sync
    }

    // Marquer comme synchronisé
    const ids = batch.map((r) => r.id);
    await markSynced(table, ids);
  }
}

function markSynced(table, ids) {
  return new Promise((resolve) => {
    const placeholders = ids.map(() => '?').join(',');
    db.transaction((tx) =>
      tx.executeSql(
        `UPDATE tableSETsynced=1WHEREidIN({table} SET synced=1 WHERE id IN (tableSETsynced=1WHEREidIN({placeholders})`,
        ids,
        () => resolve()
      )
    );
  });
}

// ============================================================
//  PULL — Cloud → Local (incrémental + résolution conflits)
// ============================================================

async function pullTable(table, userId, lastPull) {
  let query = supabase
    .from(table)
    .select('*')
    .eq('user_id', userId);

  // Sync incrémentale : seulement ce qui a changé depuis le dernier pull
  if (lastPull) {
    query = query.gt('updated_at', lastPull);
  }

  const { data, error } = await query;
  if (error) throw error;
  if (!data || data.length === 0) return;

  for (const remote of data) {
    await mergeRow(table, remote);
  }
}

/**
 * Fusionne une ligne distante dans SQLite :
 * - Si absente localement → INSERT
 * - Si présente → on compare updated_at (last-write-wins)
 *   ⚠️ Exception : si la ligne locale n'est PAS synchronisée,
 *   elle a priorité (modifications hors-ligne non encore poussées).
 */
async function mergeRow(table, remote) {
  const local = await queryLocal(`SELECT * FROM ${table} WHERE id=?`, [remote.id]);

  if (local.length === 0) {
    await insertRemote(table, remote);
    return;
  }

  const localRow = local[0];
  const localUnsynced = Number(localRow.synced) === 0;

  // Conflit : la ligne locale a été modifiée hors-ligne
  if (localUnsynced) {
    const localTime = new Date(localRow.updated_at).getTime();
    const remoteTime = new Date(remote.updated_at).getTime();

    if (localTime >= remoteTime) {
      // La version locale est plus récente → elle sera poussée au prochain push
      return;
    }
    // Sinon la version distante gagne → on écrase
  }

  await insertRemote(table, remote);
}

function insertRemote(table, row) {
  return new Promise((resolve) => {
    const cols = Object.keys(row);
    const vals = cols.map((k) => {
      const v = row[k];
      if (typeof v === 'boolean') return v ? 1 : 0;
      if (v === null || v === undefined) return null;
      if (typeof v === 'object') return JSON.stringify(v); // timestamptz etc.
      return v;
    });
    const placeholders = cols.map(() => '?').join(',');

    db.transaction((tx) =>
      tx.executeSql(
        `INSERT OR REPLACE INTO table({table} (table({cols.join(',')}, synced)
         VALUES (${placeholders}, 1)`,
        [...vals],
        () => resolve()
      )
    );
  });
}

// ============================================================
//  PURGE — Nettoyage des suppressions propagées
// ============================================================

/**
 * Les lignes marquées deleted=1 ET synced=1 ont été poussées vers le cloud.
 * On peut les purger localement après 30 jours (garde-fou anti-perte).
 */
async function purgeDeletedLocally(userId) {
  const cutoff = new Date(Date.now() - 30 * 24 * 60 * 60 * 1000).toISOString();
  for (const table of TABLES) {
    await executeLocal(
      `DELETE FROM ${table}
       WHERE user_id=? AND deleted=1 AND synced=1 AND updated_at < ?`,
      [userId, cutoff]
    );
  }
}

// ============================================================
//  HELPERS SQLite
// ============================================================

function queryLocal(sql, params = []) {
  return new Promise((resolve, reject) => {
    db.transaction((tx) =>
      tx.executeSql(
        sql, params,
        (_, { rows }) => resolve(rows._array),
        (_, err) => { reject(err); return true; }
      )
    );
  });
}

function executeLocal(sql, params = []) {
  return new Promise((resolve, reject) => {
    db.transaction((tx) =>
      tx.executeSql(
        sql, params,
        () => resolve(),
        (_, err) => { reject(err); return true; }
      )
    );
  });
}

// ============================================================
//  SYNC EN ARRIÈRE-PLAN (optionnel)
// ============================================================

let syncInterval = null;

/**
 * Démarre une sync automatique toutes les X minutes
 * quand l'app est ouverte. À appeler après le login.
 */
export function startAutoSync(userId, intervalMs = 5 * 60 * 1000) {
  stopAutoSync();
  syncInterval = setInterval(async () => {
    const state = await navigator.onLine !== undefined ? navigator.onLine : true;
    if (state) await syncAll(userId);
  }, intervalMs);
}

export function stopAutoSync() {
  if (syncInterval) clearInterval(syncInterval);
  syncInterval = null;
}

import React, { createContext, useContext, useEffect, useState } from 'react';
import { supabase } from '../config/supabase';
import { initDB } from '../services/database';
import { syncAll, startAutoSync, stopAutoSync } from '../services/syncService';
import { authenticateBiometric, isBiometricAvailable } from '../services/biometrics';
import AsyncStorage from '@react-native-async-storage/async-storage';

const AuthContext = createContext({});
export const useAuth = () => useContext(AuthContext);

// Clé pour mémoriser si l'utilisateur a activé le verrou biométrique
const BIO_LOCK_KEY = 'nkap_bio_lock_enabled';

export function AuthProvider({ children }) {
  const [session, setSession] = useState(null);
  const [loading, setLoading] = useState(true);
  const [locked, setLocked] = useState(false); // écran de verrouillage biométrique

  // ------------------------------------------------------------
  //  INITIALISATION
  // ------------------------------------------------------------
  useEffect(() => {
    let mounted = true;

    async function bootstrap() {
      // 1. Initialiser la base SQLite locale
      await initDB();

      // 2. Récupérer la session Supabase (persistée via AsyncStorage)
      const { data: { session: currentSession } } = await supabase.auth.getSession();

      if (!mounted) return;
      setSession(currentSession);

      // 3. Si l'utilisateur avait activé le verrou bio → app verrouillée au démarrage
      const bioEnabled = await AsyncStorage.getItem(BIO_LOCK_KEY);
      if (currentSession && bioEnabled === 'true') {
        setLocked(true);
      }

      setLoading(false);
    }

    bootstrap();

    // 4. Écouter les changements d'état d'authentification
    const { data: listener } = supabase.auth.onAuthStateChange(async (_event, newSession) => {
      if (!mounted) return;
      setSession(newSession);

      if (newSession?.user) {
        await ensureProfile(newSession.user);
      } else {
        stopAutoSync();
      }
    });

    return () => {
      mounted = false;
      listener?.subscription.unsubscribe();
      stopAutoSync();
    };
  }, []);

  // ------------------------------------------------------------
  //  PROFIL + SYNCHRONISATION
  // ------------------------------------------------------------
  async function ensureProfile(user) {
    try {
      // Créer/mettre à jour le profil dans le cloud
      await supabase.from('profiles').upsert({
        id: user.id,
        email: user.email,
        updated_at: new Date().toISOString(),
      });

      // Sync initiale puis auto-sync toutes les 5 minutes
      await syncAll(user.id);
      startAutoSync(user.id);
    } catch (e) {
      console.warn('[NKAP Auth] ensureProfile:', e.message);
    }
  }

  // ------------------------------------------------------------
  //  AUTHENTIFICATION
  // ------------------------------------------------------------

  /** Connexion par email + mot de passe */
  const signInEmail = async (email, password) => {
    const { data, error } = await supabase.auth.signInWithPassword({ email, password });
    if (!error && data.session) {
      // Vérifier si le verrou bio est activé → demander la biométrie
      const bioEnabled = await AsyncStorage.getItem(BIO_LOCK_KEY);
      if (bioEnabled === 'true') {
        const ok = await authenticateBiometric('Déverrouillez Nkap');
        if (!ok) {
          await supabase.auth.signOut(); // annuler la session si refus
          return { error: { message: 'Authentification biométrique requise.' } };
        }
      }
    }
    return { error };
  };

  /** Inscription par email + mot de passe */
  const signUpEmail = async (email, password) => {
    const { data, error } = await supabase.auth.signUp({ email, password });
    if (!error && data.user) {
      await supabase.from('profiles').upsert({
        id: data.user.id,
        email,
        updated_at: new Date().toISOString(),
      });
    }
    return { data, error };
  };

  /**
   * Connexion via Google OAuth.
   * ⚠️ Nécessite :
   *  - "scheme": "nkap" dans app.json
   *  - Provider Google activé dans le dashboard Supabase
   *  - expo-auth-session installé pour gérer le retour
   */
  const signInGoogle = async () => {
    try {
      const { error } = await supabase.auth.signInWithOAuth({
        provider: 'google',
        options: { redirectTo: 'nkap://auth/callback' },
      });
      return { error };
    } catch (e) {
      return { error: { message: e.message } };
    }
  };

  /** Déconnexion : arrête l'auto-sync et nettoie l'état local */
  const signOut = async () => {
    stopAutoSync();
    setLocked(false);
    await supabase.auth.signOut();
  };

  // ------------------------------------------------------------
  //  VERROU BIOMÉTRIQUE (MFA Face ID / empreinte)
  // ------------------------------------------------------------

  /** Active/désactive le verrou biométrique (appelé depuis SettingsScreen) */
  const setBioLock = async (enabled) => {
    if (enabled) {
      const available = await isBiometricAvailable();
      if (!available) {
        return { error: { message: 'Biométrie non disponible sur cet appareil.' } };
      }
      const ok = await authenticateBiometric('Confirmez pour activer le verrou');
      if (!ok) {
        return { error: { message: 'Authentification échouée.' } };
      }
    }
    await AsyncStorage.setItem(BIO_LOCK_KEY, enabled ? 'true' : 'false');
    return {};
  };

  const isBioLockEnabled = async () =>
    (await AsyncStorage.getItem(BIO_LOCK_KEY)) === 'true';

  /** Déverrouille l'app après authentification biométrique réussie */
  const unlock = async () => {
    const ok = await authenticateBiometric('Déverrouillez Nkap');
    if (ok) setLocked(false);
    return ok;
  };

  // ------------------------------------------------------------
  //  VALEUR DU CONTEXTE
  // ------------------------------------------------------------
  return (
    <AuthContext.Provider
      value={{
        session,
        loading,
        locked,
        signInEmail,
        signUpEmail,
        signInGoogle,
        signOut,
        setBioLock,
        isBioLockEnabled,
        unlock,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
}

import React, { useEffect, useState } from 'react';
import {
  View,
  Text,

export default function SettingsScreen() {
  const { session, signOut, setBioLock, isBioLockEnabled } = useAuth();

  const [bioEnabled, setBioEnabled] = useState(false);
  const [bioType, setBioType] = useState('Biométrie');
  const [bioLoading, setBioLoading] = useState(true);
  const [syncing, setSyncing] = useState(false);

  // ------------------------------------------------------------
  //  CHARGEMENT DE L'ÉTAT INITIAL
  // ------------------------------------------------------------
  useEffect(() => {
    async function load() {
      const enabled = await isBioLockEnabled();
      setBioEnabled(enabled);
      try {
        const type = await getBiometricType();
        if (type) setBioType(type);
      } catch (_) {
        // biométrie non disponible → on garde le libellé par défaut
      }
      setBioLoading(false);
    }
    load();
  }, []);

  // ------------------------------------------------------------
  //  VERROU BIOMÉTRIQUE
  // ------------------------------------------------------------
  const toggleBio = async (value) => {
    const { error } = await setBioLock(value);
    if (error) {
      Alert.alert('Erreur', error.message);
      return; // le switch reste dans son état précédent
    }
    setBioEnabled(value);
    if (value) {
      Alert.alert(
        '🔐 Verrou activé',
        `Nkap se verrouillera désormais avec ${bioType} au démarrage.`
      );
    }
  };

  // ------------------------------------------------------------
  //  SYNCHRONISATION MANUELLE
  // ------------------------------------------------------------
  const manualSync = async () => {
    setSyncing(true);
    const ok = await syncAll(session.user.id);
    setSyncing(false);
    Alert.alert(
      ok ? '✅ Synchronisation réussie' : '⚠️ Échec de synchronisation',
      ok
        ? 'Vos données locales et cloud sont à jour.'
        : 'Vérifiez votre connexion internet puis réessayez.'
    );
  };

  // ------------------------------------------------------------
  //  DÉCONNEXION (avec confirmation)
  // ------------------------------------------------------------
  const confirmSignOut = () => {
    Alert.alert('Déconnexion', 'Voulez-vous vraiment vous déconnecter ?', [
      { text: 'Annuler', style: 'cancel' },
      { text: 'Se déconnecter', style: 'destructive', onPress: signOut },
    ]);
  };

  // ------------------------------------------------------------
  //  RENDU
  // ------------------------------------------------------------
  return (
    <ScrollView style={styles.container} contentContainerStyle={{ padding: 20 }}>
      {/* ---- Profil ---- */}
      <View style={styles.card}>
        <Text style={styles.sectionTitle}>👤 Compte</Text>
        <Text style={styles.email}>{session?.user?.email || 'Utilisateur'}</Text>
      </View>

      {/* ---- Sécurité ---- */}
      <View style={styles.card}>
        <Text style={styles.sectionTitle}>🔐 Sécurité</Text>
        {bioLoading ? (
          <ActivityIndicator color="#4F46E5" style={{ marginVertical: 10 }} />
        ) : (
          <View style={styles.row}>
            <Text style={styles.rowLabel}>Déverrouillage {bioType}</Text>
            <Switch
              value={bioEnabled}
              onValueChange={toggleBio}
              trackColor={{ true: '#4F46E5' }}
            />
          </View>
        )}
        <Text style={styles.hint}>
          Protégez l'accès à vos finances avec Face ID ou votre empreinte digitale.
        </Text>
      </View>

      {/* ---- Données ---- */}
      <View style={styles.card}>
        <Text style={styles.sectionTitle}>☁️ Données</Text>
        <TouchableOpacity style={styles.btn} onPress={manualSync} disabled={syncing}>
          {syncing ? (
            <ActivityIndicator color="#fff" />
          ) : (
            <Text style={styles.btnTxt}>🔄 Synchroniser maintenant</Text>
          )}
        </TouchableOpacity>
        <Text style={styles.hint}>
          Vos données sont stockées localement et synchronisées automatiquement toutes les 5 minutes.
        </Text>
      </View>

      {/* ---- Déconnexion ---- */}
      <TouchableOpacity style={[styles.btn, styles.logoutBtn]} onPress={confirmSignOut}>
        <Text style={styles.logoutTxt}>Se déconnecter</Text>
      </TouchableOpacity>

      <Text style={styles.version}>Nkap v1.0.0</Text>
    </ScrollView>
  );
}

src/i18n/index.js — Le moteur i18n
import { I18n } from 'i18n-js';
import * as Localization from 'expo-localization';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { translations } from './translations';

const STORAGE_KEY = 'nkap_language';

const i18n = new I18n(translations);

// Langues supportées
export const SUPPORTED_LANGUAGES = [
  { code: 'fr', label: 'Français', flag: '🇫🇷' },
  { code: 'en', label: 'English', flag: '🇬🇧' },
  { code: 'es', label: 'Español', flag: '🇪🇸' },
  { code: 'ar', label: 'العربية', flag: '🇸🇦', rtl: true },
  { code: 'sw', label: 'Kiswahili', flag: '🇰🇪' },
  { code: 'it', label: 'Italiano', flag: '🇮🇹' },
  { code: 'de', label: 'Deutsch', flag: '🇩🇪' },
];

// Détection automatique de la langue du téléphone au premier lancement
export async function initLanguage() {
  const saved = await AsyncStorage.getItem(STORAGE_KEY);
  if (saved) {
    i18n.locale = saved;
  } else {
    const deviceLang = Localization.getLocales()[0]?.languageCode || 'fr';
    i18n.locale = SUPPORTED_LANGUAGES.some((l) => l.code === deviceLang)
      ? deviceLang
      : 'fr'; // fallback français
  }
  return i18n.locale;
}

// Changer de langue manuellement
export async function setLanguage(code) {
  i18n.locale = code;
  await AsyncStorage.setItem(STORAGE_KEY, code);
}

// Fonction de traduction à utiliser partout : t('login')
export function t(key) {
  return i18n.t(key);
}

// Support RTL pour l'arabe
export function isRTL() {
  return i18n.locale === 'ar';
}

export default i18n;

import React, { useEffect, useState } from 'react';
import { I18nManager } from 'react-native';
import { initLanguage, isRTL } from './src/i18n';

export default function App() {
  const [ready, setReady] = useState(false);

  useEffect(() => {
    initLanguage().then(() => {
      I18nManager.allowRTL(isRTL());   // active le RTL pour l'arabe
      setReady(true);
    });
  }, []);

  if (!ready) return null;

  return (
    <AuthProvider>
      <NavigationContainer>
        <StatusBar style="light" />
        <Root />
      </NavigationContainer>
    </AuthProvider>
  );
}

const [langModalOpen, setLangModalOpen] = useState(false);

⚠️ Points d'attention pour l'arabe (RTL)
Point	Solution
Direction du texte	I18nManager.allowRTL(true) gère automatiquement
Icônes directionnelles	Remplacer ➡️ par ⇄ ou utiliser des icônes symétriques

Polices	Ajoute une police arabe (ex: Cairo) pour un meilleur rendu
